\section{Design Principles}
\begin{frame}{Core Presentation Design Principles}
    \begin{block}{The 60-30-10 Color Rule}
        \begin{itemize}
            \item \textbf{60\% Dominant Tone}: Neutral background for eye comfort.
            \item \textbf{30% Secondary Tone}: Structural headers and container blocks.
            \item \textbf{10% Accent Tone}: High-impact highlights for critical takeaways.
        \end{itemize}
    \end{block}
    
    \begin{alertblock}{Important Contrast Warning}
        Always maintain strict luminance contrast between text and background!
    \end{alertblock}
\end{frame}
